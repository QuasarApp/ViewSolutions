56void NotificationService::questionComplete(bool accepted, int code) {
57
58 if (_listeners.contains(code)) {
59 auto listener = _listeners.value(code);
60 listener(accepted);
61 _listeners.remove(code);
62 }
63
64 emit questionCompleted(accepted, code);

76int NotificationService::setQuestion(QObject *listenerObject,
77 const QString &listenerMethod,
78 const QString &title,
79 const QString &text,
80 const QString &img,
81 int code) {
82
83 Listener listener = [listenerObject, listenerMethod](bool accept) {
84
85 const QByteArray stringData = listenerMethod.toLatin1();
86 char method[100];
87 method[qMin(99, stringData.size())] = '\0';
88 std::copy(stringData.constBegin(), stringData.constBegin() + qMin(99, stringData.size()), method);
89
90 QMetaObject::invokeMethod(listenerObject, method,
91 Qt::QueuedConnection, Q_ARG(bool, accept));
92 };
93
94 return setQuestion(listener, NotificationData(title, text, img, code));
95}
